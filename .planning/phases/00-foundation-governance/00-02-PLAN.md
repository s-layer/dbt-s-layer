```python
"""Environment diagnostics for dbt-s-layer.

Phase 0 scope (per CONTEXT.md):
- Detects `dbtf` (dbt Fusion) binary on PATH via `shutil.which`
- Reports installed `metricflow` version via `importlib.metadata.version`

Deferred to Phase 1: semantic_manifest.json freshness check.
Deferred to Phase 3: output directory writability check.
"""

from __future__ import annotations

import shutil
import sys
from importlib.metadata import PackageNotFoundError, version

FUSION_INSTALL_URL = "https://docs.getdbt.com/docs/fusion/install-fusion"


def run_doctor() -> int:
    """Report environment diagnostics and return an exit code.

    Exit codes:
        0 — all checks passed (dbtf on PATH AND metricflow importable)
        1 — dbtf binary missing (blocking; user must install Fusion)
        2 — metricflow package not installed (dbt-s-layer install is broken)

    When multiple checks fail, the higher-severity code wins (2 > 1 > 0).
    """
    exit_code = 0

    dbtf_path = shutil.which("dbtf")
    if dbtf_path is None:
        print("ERROR: `dbtf` (dbt Fusion) not found on PATH.", file=sys.stdout)
        print(f"Install Fusion: {FUSION_INSTALL_URL}", file=sys.stdout)
        exit_code = max(exit_code, 1)
    else:
        print(f"OK  dbtf found at: {dbtf_path}", file=sys.stdout)

    try:
        mf_version = version("metricflow")
        print(f"OK  metricflow version: {mf_version}", file=sys.stdout)
    except PackageNotFoundError:
        print("ERROR: metricflow is not installed (dbt-s-layer install is broken).", file=sys.stdout)
        exit_code = max(exit_code, 2)

    return exit_code
```

Notes:
- `print(..., file=sys.stdout)` is explicit so CliRunner captures in `result.stdout` (Typer's CliRunner uses `mix_stderr=False` by default in recent versions).
- `from __future__ import annotations` is used for basedpyright-strict-compatible type hints.
- Exit-code precedence is `max(exit_code, new_value)` so if both checks fail we return `2`.

```python
"""dbt-s-layer command-line interface."""

from __future__ import annotations

import typer

app = typer.Typer(
    name="dbt-s-layer",
    help="Code-generate dbt models from the dbt semantic layer.",
    no_args_is_help=True,
    add_completion=False,
)


@app.command()
def doctor() -> None:
    """Report environment diagnostics (Fusion binary, metricflow version)."""
    from dbt_s_layer._doctor import run_doctor

    raise typer.Exit(code=run_doctor())


if __name__ == "__main__":
    app()
```

Notes:
- `no_args_is_help=True` ensures `dbt-s-layer` (with no args) prints help instead of erroring — Typer idiom
- `add_completion=False` skips the `--install-completion` / `--show-completion` global options since CONTEXT.md scopes Phase 0 tightly to the `doctor` subcommand
- The `run_doctor` import is lazy (inside the command) so `--help` is fast and doesn't trigger `importlib.metadata` calls
